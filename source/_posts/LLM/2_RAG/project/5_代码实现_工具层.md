# 1. 模块设计
## 1.1 模块定位
- 已解决：
	- 存储结构
	- 文档入库
	- 单次检索
	- RAG Pipeline
- 当前：将 **RAG Pipeline**打包为一个工具。供后续agent使用。同时这里也稍微增加一个测试工具：天气工具。
## 1.2 模块流程设计
```
	                Agent
	                  ↓
	               调用 Tool
               ↙              ↘
          Knowledge Tool     Weather Tool
               ↓                 ↓
          RAG Pipeline       高德天气 API
               ↓                 ↓
          RAG Result         Weather Result
               ↓                 ↓
         处理路由与证据       格式化天气数据
               ↓                 ↓
               └────────┬────────┘
                        ↓
                   Tool Result
                        ↓
                      Agent
                        ↓
                    最终回答
```

## 1.3 文件结构设计
```python
backend/
└── tools/
    ├── knowledge.py        # 知识库工具：封装 RAG Pipeline
    ├── weather.py          # 天气工具：封装高德天气 API
    └── __init__.py         # 统一导出 Agent Tools
```

# 2. 模块实现
## 2.1 knowledge.py
该文件实现rag工具，调用rag工具时候需要一些信息，例如针对当前query，我之前有没有调用过rag工具，如果已经调用过了，我就不让他再用了（单次query只允许调用一次rag工具），rag工具检索到信息之后如何返回给agent？对此我们用一个上下文对象来在 Agent、Tool 和 RAG 之间共享临时状态，用户每次发起一个query我们就会新建一个上下文对象，作为传递信息的数据包。使用流程上，可以用闭包的形式创建，用户每次发起一个query，我们就创建一次工具，同时创建好这个上下文对象，当调用工具的时候就只需要传入query就行了，执行工具时自动更新这个闭包里面的上下文对象。
1. 定义工具创建函数，通过闭包的形式传入上下文对象。（类似创建了一个类和属性）
2. 该函数里面再定义工具函数：
	1. `ctx.acquire_knowledge_tool_slot()`：通过上下文对象检查本轮 Knowledge Tool 调用次数，超过限制直接返回。
	2. `run_rag_graph()`：调用前面实现的 RAG Pipeline，得到 `docs`、`rag_trace`、`hitl_resume_state`等信息。
	3. `ctx.store_rag_trace()`：讲执行轨迹和 HITL 恢复状态保存到上下文，供后续 Chat Service 使用。
	4. 根据 RAG 路由 返回不同结果：
		- `answer` 返回格式化 Evidence；
		- `clarify / scope_select` 返回用户补充提示；
		- `no_knowledge` 返回无知识提示。
3. 闭包返回工具函数。
```python
from langchain_core.tools import tool

from backend.chat.request_context import ChatRequestContext


def make_search_knowledge_base(ctx: ChatRequestContext):
    @tool("search_knowledge_base") # 注册一个工具
    def search_knowledge_base(query: str) -> str:
        """Search for information in the knowledge base using hybrid retrieval (dense + sparse vectors)."""
         # 1. 检查单轮调用次数
        if not ctx.acquire_knowledge_tool_slot():
            return (
                "TOOL_CALL_LIMIT_REACHED: search_knowledge_base has already been called once in this turn. "
                "Use the existing retrieval result and provide the final answer directly."
            )

        # Delayed import keeps tests and lightweight imports away from RAG/embedding startup.
        # 2. 延迟导入并执行 RAG Pipeline
        from backend.rag.pipeline import run_rag_graph

        rag_result = run_rag_graph(query, ctx)

        # 3. 提取 RAG 返回的数据
        docs = rag_result.get("docs", []) if isinstance(rag_result, dict) else []
        rag_trace = rag_result.get("rag_trace", {}) if isinstance(rag_result, dict) else {}
        hitl_resume_state = (
            rag_result.get("hitl_resume_state")
            if isinstance(rag_result, dict)
            else None
        )

        # 4. 保存 RAG Trace 和 HITL 恢复状态
        ctx.store_rag_trace(rag_trace, hitl_resume_state)

         # 5. 根据 RAG 路由处理结果
        status = rag_trace.get("retrieval_status") if isinstance(rag_trace, dict) else None
        route = rag_trace.get("route") if isinstance(rag_trace, dict) else None

        # 5.1 缺少关键条件，需要用户澄清
        if status == "needs_clarification" or route == "clarify":
            prompt = rag_trace.get("hitl_prompt") or "I found related knowledge, but need one more detail before answering."
            return f"NEEDS_CLARIFICATION: {prompt}"
        
        # 5.2 存在多个候选方向，需要用户选择
        if status == "needs_scope_selection" or route == "scope_select":
            prompt = rag_trace.get("hitl_prompt") or "I found multiple related knowledge-base directions. Ask the user to choose one."
            options = rag_trace.get("hitl_options") or []
            if options:
                prompt = f"{prompt}\nOptions: " + "; ".join(str(item) for item in options)
            return f"NEEDS_SCOPE_SELECTION: {prompt}"

        # 5.3 没有可靠 Evidence
        if status == "no_knowledge" or route == "no_knowledge":
            return "NO_KNOWLEDGE: No reliable relevant documents were found in the knowledge base."

        if not docs:
            return "No relevant documents found in the knowledge base."

        # 5.4 将 Evidence 格式化为 Agent 可使用的文本
        formatted = []
        for i, result in enumerate(docs, 1):
            source = result.get("filename", "Unknown")
            page = result.get("page_number", "N/A")
            text = result.get("text", "")
            formatted.append(f"[{i}] {source} (Page {page}):\n{text}")

        return "Retrieved Chunks:\n" + "\n\n---\n\n".join(formatted)

    # 返回 LangChain Tool 对象，供 Agent 注册
    return search_knowledge_base
```
## 2.2 weather.py
提供一个测试工具：天气获取。该工具不需要额外的上下文信息，只需要针对输入进行查询就行了。因此不需要闭包的形式，这里提供两个函数，一个业务函数，一个接口函数：
```python
import os
from typing import Optional

import requests

try:
    from langchain_core.tools import tool
except ImportError:
    from langchain_core.tools import tool

AMAP_WEATHER_API = os.getenv("AMAP_WEATHER_API")
AMAP_API_KEY = os.getenv("AMAP_API_KEY")


def get_current_weather(location: str, extensions: Optional[str] = "base") -> str:
    """获取天气信息"""
    if not location:
        return "location参数不能为空"
    if extensions not in ("base", "all"):
        return "extensions参数错误，请输入base或all"

    if not AMAP_WEATHER_API or not AMAP_API_KEY:
        return "天气服务未配置（缺少 AMAP_WEATHER_API 或 AMAP_API_KEY）"

    params = {
        "key": AMAP_API_KEY,
        "city": location,
        "extensions": extensions,
        "output": "json",
    }

    try:
        resp = requests.get(AMAP_WEATHER_API, params=params, timeout=10)
        resp.raise_for_status()
        data = resp.json()
        if data.get("status") != "1":
            return f"查询失败：{data.get('info', '未知错误')}"

        if extensions == "base":
            lives = data.get("lives", [])
            if not lives:
                return f"未查询到 {location} 的天气数据"
            w = lives[0]
            return (
                f"【{w.get('city', location)} 实时天气】\n"
                f"天气状况：{w.get('weather', '未知')}\n"
                f"温度：{w.get('temperature', '未知')}℃\n"
                f"湿度：{w.get('humidity', '未知')}%\n"
                f"风向：{w.get('winddirection', '未知')}\n"
                f"风力：{w.get('windpower', '未知')}级\n"
                f"更新时间：{w.get('reporttime', '未知')}"
            )

        forecasts = data.get("forecasts", [])
        if not forecasts:
            return f"未查询到 {location} 的天气预报数据"
        f0 = forecasts[0]
        out = [f"【{f0.get('city', location)} 天气预报】", f"更新时间：{f0.get('reporttime', '未知')}", ""]
        today = (f0.get("casts") or [])[0] if f0.get("casts") else {}
        out += [
            "今日天气：",
            f"  白天：{today.get('dayweather','未知')}",
            f"  夜间：{today.get('nightweather','未知')}",
            f"  气温：{today.get('nighttemp','未知')}~{today.get('daytemp','未知')}℃",
        ]
        return "\n".join(out)

    except requests.exceptions.Timeout:
        return "错误：请求天气服务超时"
    except requests.exceptions.RequestException as e:
        return f"错误：天气服务请求失败 - {e}"
    except Exception as e:
        return f"错误：解析天气数据失败 - {e}"


@tool("get_current_weather")
def get_current_weather_tool(location: str, extensions: Optional[str] = "base") -> str:
    """Get current weather for a city in China."""
    return get_current_weather(location, extensions)
```
