## 1.2 模块流程设计
```
                     用户发送 Query
                           ↓
                    加载当前会话数据
                    ├── 历史聊天记录
                    ├── 长期记忆摘要
                    └── 待处理的 HITL （判断当前query是不是对HITL的补充信息）
                           ↓
                    创建本轮请求 ctx（记录本次query执行过程的临时状态）
                           ↓
                 判断是否需要恢复 HITL
                    ↙             ↘
                  否               是
                  ↓                ↓
            构建 Agent 上下文    结合用户补充信息
            ├── 历史消息        拼接恢复 RAG 检索
            ├── 长期记忆             ↓
            └── 当前 Query       重新判断证据
                  ↓                ↓
              创建 Agent           │
                  ↓                │
             LLM 推理与决策         │
             ├── 直接回答           │
             ├── Knowledge Tool    │
             │      └── RAG        │
             └── Weather Tool      │
                  ↓                │
             获取工具执行结果        │
                  ↓                │
             LLM 生成回答           │
                  └────────┬───────┘
                           ↓
                    检查本轮结果
                   ↙          ↘
               正常回答       需要 HITL
                  ↓              ↓
             按条件更新记忆   生成澄清问题
                  ↓              ↓
                  │         保存待恢复 HITL 状态
                  ↓              ↓
             保存用户消息、AI 消息
                  和会话信息
                       ↓
                  关闭本轮 ctx
                       ↓
                   返回给用户
                   ↙        ↘
```

# 2. 模块实现
## 2.1 对话流程编排
先提供一个同步的对话编排总体流程：
1. 从redis或数据库加载会话历史，判断是否需要 HITL 恢复。
2. 创建本轮 `ChatRequestContext`，把原始用户消息**先保存一次**。
3. 如果存在有效 `resume_state`，走 **RAG HITL 恢复**；否则创建 Agent，用最近 6 条历史消息和笔记推理。
4. 从 RAG 结果或 `ctx.take_rag_trace()` 获取执行轨迹，判断本轮是否再次需要澄清。
5. 更新metadata: 标题、HITL、长期记忆
6. 保存最终 AIMessage 与本轮 Trace。
> ???这里面有三个待优化的问题：
> 1. 一次query要做两次全量的save，可以优化成增量insert
> 2. 历史消息的rag trace会丢失
> 3. 无法判断用户输入的是一个补充信息还是别的问题
```python
def chat_with_agent(
    user_text: str,
    user_id: str = "default_user",
    session_id: str = "default_session",
):
    '''
    同步对话入口：负责从加载 Session 到保存最终回答的完整单轮编排。
    '''
    # 1. 从redis或数据库加载会话历史，判断是否需要 HITL 恢复
    messages, metadata = storage.load_with_meta(user_id, session_id)
    persistent_note = metadata.get("persistent_note", "") # 之前保存的长期记忆摘要，随着每轮对话滚动更新
    is_first_message = len(messages) == 0 # 当前 Query 是否为这个 Session 的第一条消息
    # 从元数据里恢复待处理 HITL，识别无效/有效状态。
    stored_pending_hitl = metadata.get(PENDING_HITL_KEY) # 上一轮的hitl状态
    pending_hitl = _current_pending_hitl(stored_pending_hitl) # 校验
    invalid_pending_hitl = stored_pending_hitl is not None and pending_hitl is None # 数据库中有 HITL 状态，但校验失败
    is_hitl_resume = isinstance(pending_hitl, dict) # 本轮需要尝试恢复
    resume_state = _pending_resume_state(pending_hitl) # 提取 RAG Pipeline 恢复所需的快照
    # 如果是 HITL 补充，就组合原问题与补充信息
    effective_user_text = (
        _build_hitl_resume_query(pending_hitl, user_text) #usertext是hitl提示之后用户输入的补充信息
        if is_hitl_resume
        else user_text
    )
    hitl_answers = _existing_hitl_answers(pending_hitl)  # 人为补充信息，可能要多轮补充，因此列表累积usertext
    if is_hitl_resume: # 本轮也是一次补充，将其加入补充信息列表
        hitl_answers = [*hitl_answers, user_text]
    original_question = ( # 记录原始问题，供之后可能再次触发 HITL 时使用
        pending_hitl.get("original_question")
        if is_hitl_resume
        else user_text
    )

    # 2. 创建 ctx，先保存用户消息
    ctx = ChatRequestContext.for_sync(user_id=user_id, session_id=session_id) # 创建本次 Query 专属的同步运行上下文
    ctx.reset_knowledge_tool_budget()

    try:
        messages.append(HumanMessage(content=user_text)) # 当前用户输入追加到会话历史
        storage.save(user_id, session_id, messages) # 保存完整消息列表

    # 3. 分支A: 上一轮 HITL 恢复
        if is_hitl_resume and resume_state:
            ## 3.1 使用保存的hitl状态+当前用户补充信息， 直接恢复之前的rag pipeline
            rag_result = _resume_rag_from_hitl_sync(pending_hitl, user_text, ctx)
            ## 3.2 提取恢复之后的rag trace
            rag_trace = normalize_rag_trace(
                rag_result.get("rag_trace") if isinstance(rag_result, dict) else None
            )
            next_pending_hitl = None
            ## 3.3 恢复之后仍要补充信息
            if _is_hitl_trace(rag_trace):
                next_pending_hitl = _build_pending_hitl(
                    rag_trace,
                    original_question or user_text,
                    previous_answers=hitl_answers,
                    resume_state=rag_result.get("hitl_resume_state"),
                )
                response_content = _format_hitl_message(
                    next_pending_hitl["prompt"],
                    next_pending_hitl["options"],
                )
            ## 3.4 恢复之后不需要补充信息了， 尝试根据证据回答
            else:
                response_content = _answer_resumed_rag_sync(pending_hitl, user_text, rag_result)
    # 4. 分支B: 普通对话，创建 Agent 并执行
        else:
            ## 4.1 创建绑定当前 ctx 的 Agent
            request_agent = create_agent_for_request(ctx)
            ## 4.2 构建Agent输入上下文
            context_messages = _build_context_messages(messages[:-1], persistent_note, effective_user_text)
            ## 4.3 执行Agent
            result = request_agent.invoke(
                {"messages": context_messages},
                config={"recursion_limit": 8},
            )
            ## 4.4 从 Agent 返回结果中提取回答
            response_content = ""
            if isinstance(result, dict):
                if "output" in result:
                    response_content = result["output"]
                elif "messages" in result and result["messages"]:
                    msg = result["messages"][-1]
                    response_content = getattr(msg, "content", str(msg))
                else:
                    response_content = str(result)
            elif hasattr(result, "content"):
                response_content = result.content
            else:
                response_content = str(result)
            ## 4.5. 取出 Knowledge Tool 存储在 ctx 中的 RAG 信息
            stored_trace = ctx.take_rag_trace()
            rag_trace = normalize_rag_trace(stored_trace.get("rag_trace") if stored_trace else None)
            resume_state_from_trace = stored_trace.get("hitl_resume_state") if stored_trace else None
            next_pending_hitl = None
            ## 4.6 判断本轮是否触发 HITL
            if _is_hitl_trace(rag_trace):
                next_pending_hitl = _build_pending_hitl(
                    rag_trace,
                    original_question or user_text,
                    previous_answers=hitl_answers,
                    resume_state=resume_state_from_trace,
                )
                response_content = _format_hitl_message(
                    next_pending_hitl["prompt"],
                    next_pending_hitl["options"],
                )

        # 5. 更新 metadata：标题、HITL、长期记忆
        save_meta = dict(metadata)
        if invalid_pending_hitl: #清理之前保存的无效 HITL 状态
            save_meta[PENDING_HITL_KEY] = None
        if is_first_message: #如果是 Session 的第一条消息，生成标题
            save_meta["title"] = generate_session_title(user_text)
        if next_pending_hitl: #本轮再次触发 HITL，保存新的待恢复任务
            save_meta[PENDING_HITL_KEY] = next_pending_hitl
        else: # 本轮没有待处理 HITL
            if is_hitl_resume: ## 如果完成了上一轮 HITL，清除待恢复状态
                save_meta[PENDING_HITL_KEY] = None
            # # 检查是否需要更新长期记忆
            if _should_update_persistent_note(messages, persistent_note):
                save_meta["persistent_note"] = _update_persistent_note_sync(
                    persistent_note,
                    effective_user_text,
                    response_content,
                    history_messages=messages[:-1] if not persistent_note else None,
                )

        # 6. 保存 AI 消息和本轮 RAG Trace
        messages.append(AIMessage(content=response_content))
        # 为全部历史消息创建附加数据列表，只给最后一条 AI 消息附加当前 RAG Trace
        extra_message_data = [None] * (len(messages) - 1) + [{"rag_trace": rag_trace}]  # 这里也有待优化，他会删掉之前对话历史的rag trace???
        storage.save( ## 完整保存历史消息及 Session metadata
            ## 本函数执行了两次save，
            ## 如果只在 Agent 完成后保存，Agent 中途失败可能导致用户输入也没有被记录。
            ## 因此需要两次，但是本函数的保存每次保存都全量重写所有历史消息，可以优化成增量insert???
            user_id, 
            session_id,
            messages,
            metadata=save_meta,
            extra_message_data=extra_message_data,
        )

        return {
            "response": response_content,
            "rag_trace": rag_trace,
        }
    finally:
        ctx.close()
        
def generate_session_title(user_text: str) -> str:
    compact_title = " ".join(user_text.split()).strip(" \t\r\n。！？!?，,；;：:")
    return compact_title[:16] or "新会话"
```
## 2.2 Agent 输入上下文构建
我们先走无HITL的分支，通过两个函数构建Agent的输入上下文（`ChatRequestContext`是管理本轮工具调用次数、RAG Trace、流式事件等临时状态的上下文；我们这里构建的是组织长期记忆、历史消息和当前 Query，作为 LLM 的输入的上下文；二者都是每个query创建一次）
1. 截取最近的n条消息作为短期记忆
2. 构造Agent上下文：
	- 长期记忆
	- 短期记忆
	- 最新query
```python
def _build_context_messages(
    messages: list,# 当前 Session 的历史消息（不包含当前 Query）
    persistent_note: str,# 当前 Session 的长期记忆摘要
    user_text: str,# 本轮用户输入的 Query
) -> list:
    '''
    构建本轮 Agent 的输入上下文
    将长期记忆摘要、最近 6 条历史消息和当前 Query 组合成 LangChain Message 列表，提供给 Agent 推理。
    '''
    # 1. 截取最近 6 条历史消息作为短期记忆
    short_term = messages[-CONTEXT_WINDOW_MESSAGES:] if len(messages) > CONTEXT_WINDOW_MESSAGES else messages
    context_messages: list = []
    # 2. 存入：持久化记忆、短期记忆、当前query
    if persistent_note:
        context_messages.append(
            SystemMessage(
                content=(
                    "【对话持久化笔记（你的工作记忆）】\n"
                    f"{persistent_note}\n"
                    "请参考以上笔记保持对话连贯性，避免重复回答已解决的问题。"
                )
            )
        )
    context_messages.extend(short_term)
    context_messages.append(HumanMessage(content=user_text))
    return context_messages
```

## 2.3 HITL状态管理
#### 2.3.1 历史 HITL 状态读取与恢复准备

作用： 对话开始时，检查上一轮是否存在待处理的 HITL，并构建恢复信息。

| 顺序  | 函数                           | 作用                                    |
| --- | ---------------------------- | ------------------------------------- |
| 1   | `_current_pending_hitl()`    | 校验从 Session metadata 读取的 Pending HITL |
| 2   | `_pending_resume_state()`    | 提取 RAG Pipeline 恢复快照                  |
| 3   | `_existing_hitl_answers()`   | 获取用户此前累计的补充信息                         |
| 4   | `_build_hitl_resume_query()` | 组合原始问题、历史补充和当前补充信息                    |
```python

def _current_pending_hitl(value: dict | None) -> dict | None:
    '''
    校验从 Session metadata 中读取的 pending_hitl，
    如果数据合法，则返回标准化后的 HITL 状态；
    否则返回 None。
    '''
    if not isinstance(value, dict):
        return None
    try:
        #使用 PendingHitlState Schema 校验字段和类型
        # model_validate()：将字典转换为 Pydantic 对象
        # model_dump()：将校验后的对象重新转换为字典
        return PendingHitlState.model_validate(value).model_dump()
    except ValueError:
        return None

def _pending_resume_state(pending_hitl: dict | None) -> dict | None:
    # 提取 RAG Pipeline 恢复快照
    if not isinstance(pending_hitl, dict):
        return None
    resume_state = pending_hitl.get("resume_state")
    return dict(resume_state) if isinstance(resume_state, dict) else None

def _existing_hitl_answers(pending_hitl: dict | None) -> list[str]:
    # 提取用户之前已经提供的补充信息，并过滤空字符串。
    if not isinstance(pending_hitl, dict):
        return []
    answers = pending_hitl.get("answers") or []
    if not isinstance(answers, list):
        return []
    return [str(answer).strip() for answer in answers if str(answer).strip()]
    
def _build_hitl_resume_query(pending_hitl: dict, user_text: str) -> str:
    # 组合原始问题、历史补充和当前补充信息
    original_question = pending_hitl.get("original_question") or ""
    prompt = pending_hitl.get("prompt") or ""
    previous_answers = _existing_hitl_answers(pending_hitl)

    lines = [
        "这是上一轮 RAG 流程中 HITL 澄清后的继续请求。",
        "不要把用户补充单独当成新问题；请回到原始问题继续完成回答。",
        f"原始问题：{original_question}",
    ]
    if prompt:
        lines.append(f"HITL 问题：{prompt}")
    if previous_answers:
        lines.append("此前用户已补充：")
        lines.extend(f"- {answer}" for answer in previous_answers)
    lines.extend([
        f"本轮用户补充：{user_text}",
        "请基于以上补充形成完整查询，并按原来的 Agent/RAG 流程继续。",
    ])
    return "\n".join(lines)
```
#### 2.3.2 新 HITL 的识别与信息提取

作用： Agent/RAG 执行结束后，判断是否需要用户澄清，并提取 HITL 信息。

| 顺序  | 函数                           | 作用                     |
| --- | ---------------------------- | ---------------------- |
| 1   | `_is_hitl_trace()`           | 判断 RAG Trace 是否触发 HITL |
| 2   | `_hitl_route_from_trace()`   | 判断 HITL 的类型            |
| 3   | `_hitl_prompt_from_trace()`  | 获取 HITL 提示问题           |
| 4   | `_hitl_options_from_trace()` | 获取 HITL 候选选项           |
- 最后两个函数并不是分别专门针对clarify和select类型的。`clarify` 是补充问题缺失的条件，`scope_select` 是选择检索范围；`hitl_prompt` 提供澄清问题，`hitl_options` 提供可选答案，两种 HITL 都可以使用。
```python

def _is_hitl_trace(rag_trace: dict | None) -> bool:
    # 判断 RAG Trace 是否触发 HITL
    if not isinstance(rag_trace, dict):
        return False
    status = rag_trace.get("retrieval_status")
    route = rag_trace.get("route")
    return status in HITL_STATUSES or route in HITL_ROUTES
    
def _hitl_route_from_trace(rag_trace: dict) -> str:
    # 判断 HITL 的类型
    status = rag_trace.get("retrieval_status")
    route = rag_trace.get("route")
    if status == "needs_scope_selection" or route == "scope_select":
        return "scope_select"
    return "clarify"
    
def _hitl_prompt_from_trace(rag_trace: dict) -> str:
    # 获取 HITL 提示问题
    prompt = (rag_trace.get("hitl_prompt") or "").strip()
    if prompt:
        return prompt
    # 返回默认提示语
    route = _hitl_route_from_trace(rag_trace)
    if route == "scope_select":
        return "我找到了多个可能相关的知识库方向，请选择你想继续查询的方向。"
    return "我找到了相关知识，但还缺少一个关键信息，请补充后我继续查询。"
    
def _hitl_options_from_trace(rag_trace: dict) -> list[str]:
    # 获取 HITL 候选选项
    options = rag_trace.get("hitl_options") or []
    if not isinstance(options, list):
        return []
    return [str(option).strip() for option in options if str(option).strip()]
```
#### 2.3.3 HITL 状态构建与消息生成

作用： 将新的 HITL 信息保存为跨轮状态，同时生成用户能看到的澄清消息。

| 顺序  | 函数                       | 作用                        |
| --- | ------------------------ | ------------------------- |
| 1   | `_build_pending_hitl()`  | 整合 HITL 信息，构建待恢复的 HITL 状态 |
| 2   | `_format_hitl_message()` | 将澄清问题、选项拼接为 AI 回复（字符串）    |
```python

def _build_pending_hitl(
    rag_trace: dict,
    original_question: str,
    previous_answers: list[str] | None = None,
    resume_state: dict | None = None,
) -> dict:
    '''
    当 RAG Pipeline 触发 HITL 时，
    将提示问题、候选选项、历史补充信息和 RAG 恢复快照封装成 PendingHitlState，
    供下一轮对话恢复使用。
    '''
    prompt = _hitl_prompt_from_trace(rag_trace)
    options = _hitl_options_from_trace(rag_trace)
    route = _hitl_route_from_trace(rag_trace)
    return PendingHitlState(
        id=uuid4().hex, # 生成本次 HITL 的唯一 ID
        original_question=original_question,
        prompt=prompt,
        options=options,
        route=route,
        retrieval_status=(
            "needs_scope_selection" if route == "scope_select" else "needs_clarification"
        ),
        answers=previous_answers or [],
        resume_state=resume_state,
        created_at=datetime.now(timezone.utc).isoformat(),
    ).model_dump()
    
def _format_hitl_message(prompt: str, options: list[str] | None = None) -> str:
    # 将澄清问题、选项拼接为 AI 回复（字符串）
    clean_prompt = prompt.strip()
    clean_options = [item for item in (options or []) if item]
    if not clean_options:
        return clean_prompt
    option_lines = "\n".join(f"- {item}" for item in clean_options)
    return f"{clean_prompt}\n\n可选方向：\n{option_lines}"
```

## 2.4 HITL 恢复与回答生成

作用： 用户补充信息后，恢复上一轮 RAG Pipeline，并根据新的检索证据生成回答。

| 顺序  | 函数                                | 作用                                                  |
| --- | --------------------------------- | --------------------------------------------------- |
| 1   | `_resume_rag_from_hitl_sync()`    | 根据 HITL 恢复快照和用户补充信息，重新执行 RAG                        |
| 2   | `_format_retrieved_chunks()`      | 将检索到的 Evidence 格式化为带来源编号的文本，用于输入llm                 |
| 3   | `_build_resume_answer_messages()` | 构建恢复后用于 LLM 回答的 Prompt                              |
| 4   | `_no_knowledge_response()`        | 生成知识库证据不足时的回复                                       |
| 5   | `_extract_ai_content()`           | 从 LangChain Message 中提取纯文本字符串，统一处理不同的 `content` 格式。 |
| 6   | `_answer_resumed_rag_sync()`      | 综合检索证据和 Prompt，生成最终回答                               |
```python

def _resume_rag_from_hitl_sync(pending_hitl: dict, user_answer: str, ctx: ChatRequestContext) -> dict:
    # 根据 HITL 恢复快照和用户补充信息，重新执行 RAG
    from backend.rag.pipeline import resume_rag_from_hitl

    # 1. 从 pending_hitl 中提取 RAG 恢复快照
    resume_state = _pending_resume_state(pending_hitl)
    if not resume_state:
        return {}
     # 3. 将恢复快照、用户补充信息和请求上下文交给 RAG Pipeline
    return resume_rag_from_hitl(resume_state, user_answer, ctx)
    
def _format_retrieved_chunks(docs: list[dict]) -> str:
    # 将检索到的 Evidence 格式化为带来源编号的文本，用于输入llm
    formatted = []
    for i, result in enumerate(docs, 1):
        source = result.get("filename", "Unknown")
        page = result.get("page_number", "N/A")
        text = result.get("text", "")
        formatted.append(f"[{i}] {source} (Page {page}):\n{text}")
    return "\n\n---\n\n".join(formatted)
    
def _build_resume_answer_messages(
    pending_hitl: dict,
    user_answer: str,
    docs: list[dict],
) -> list:
    # 构建恢复后用于 LLM 回答的 Prompt
    original_question = pending_hitl.get("original_question") or ""
    prompt = pending_hitl.get("prompt") or ""
    context = _format_retrieved_chunks(docs)
    system = SystemMessage(
        content=(
            "You are a helpful knowledge-base assistant. "
            "Answer the user's original question using only the retrieved chunks. "
            "You MUST cite source chunks inline with [1], [2], etc. "
            "If the chunks are insufficient, say so honestly. "
            "Do not mention internal HITL or RAG implementation details."
        )
    )
    human = HumanMessage(
        content=(
            "原始问题：\n"
            f"{original_question}\n\n"
            "HITL 补充问题：\n"
            f"{prompt}\n\n"
            "用户补充：\n"
            f"{user_answer}\n\n"
            "检索片段：\n"
            f"{context}\n\n"
            "请基于检索片段回答原始问题，并使用 [1]、[2] 这样的引用。"
        )
    )
    return [system, human]
    
def _no_knowledge_response() -> str:
    return "知识库中没有找到可靠的相关信息，暂时无法基于知识库回答这个问题。"
    
def _extract_ai_content(msg) -> str:
    # 从 LangChain Message 中提取纯文本字符串，统一处理不同的 `content` 格式。
    content = getattr(msg, "content", "")
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        text = ""
        for block in content:
            if isinstance(block, str):
                text += block
            elif isinstance(block, dict) and block.get("type") == "text":
                text += block.get("text", "")
        return text
    return str(content or "")
    
def _answer_resumed_rag_sync(pending_hitl: dict, user_answer: str, rag_result: dict) -> str:
    # 综合检索证据和 Prompt，生成最终回答
    docs = rag_result.get("docs") or []
    trace = rag_result.get("rag_trace") or {}
    status = rag_result.get("retrieval_status") or trace.get("retrieval_status")
    route = rag_result.get("route") or trace.get("route")
    if status == "no_knowledge" or route == "no_knowledge" or not docs:
        return _no_knowledge_response()
    res = model.invoke(_build_resume_answer_messages(pending_hitl, user_answer, docs))
    return _extract_ai_content(res)
```
## 2.5 长期记忆管理

作用： 判断是否需要更新 Session 摘要，并使用 `fast_model` 生成新的滚动摘要。

| 1   | `_should_update_persistent_note()` | 判断是否需要更新长期记忆         |
| 2   | `_update_persistent_note_sync()`   | 使用旧摘要和最新问答生成新摘要      |
| 3   | `update_persistent_note()`         | 将记忆更新封装为异步调用，供流式流程使用 |
```python

def _should_update_persistent_note(messages: list, current_note: str) -> bool:
    """Only pay for note maintenance once short-term context actually starts trimming."""
    # 判断是否需要更新长期记忆, 已有长期记忆或者历史消息长度达到阈值
    return bool(current_note) or len(messages) > CONTEXT_WINDOW_MESSAGES
    
def _update_persistent_note_sync(
    current_note: str,
    user_text: str,
    ai_response: str,
    *,
    history_messages: list | None = None,
) -> str:
    # 使用旧摘要和最新问答生成新摘要
    try:
        history_text = ""
        if history_messages:
            history_lines = []
            for message in history_messages:
                role = "用户" if isinstance(message, HumanMessage) else "AI"
                history_lines.append(f"{role}：{_extract_ai_content(message)}")
            history_text = (
                "\n\n▼ 首次建立笔记时需要一并概括的此前对话：\n"
                + "\n".join(history_lines)
                + "\n\n"
            )
        prompt = (
            "你是一个【Context Manager Agent】(上下文管理器)，负责维护多轮对话中的「持久化笔记」。\n"
            "笔记是模型在有限上下文窗口下的长效工作记忆，记录已解决的问题与关键事实。\n\n"
            "更新规则：\n"
            "1. 将新信息与现有笔记智能合并，不要简单拼接。\n"
            "2. 过滤噪音，控制在 500 字以内，用简明条目输出。\n"
            "3. 若信息冲突，保留最可靠或最新版本。\n\n"
            f"▼ 现有笔记：\n{current_note if current_note else '无'}\n\n"
            f"{history_text}"
            f"▼ 最新一轮对话：\n用户：{user_text}\nAI：{ai_response}\n\n"
            "请直接输出更新后的笔记（纯文本，不要解释或 Markdown 代码块）："
        )
        res = fast_model.invoke([HumanMessage(content=prompt)])
        return (res.content or "").strip()
    except Exception as e:
        print(f"Context Manager Error: {e}")
        return current_note
        
async def update_persistent_note(
    current_note: str,
    user_text: str,
    ai_response: str,
    history_messages: list | None = None,
) -> str:
    loop = asyncio.get_running_loop()# 获取当前正在运行的 asyncio 事件循环
    return await loop.run_in_executor(# 在线程池中执行同步函数
        None,
        lambda: _update_persistent_note_sync(
            current_note,
            user_text,
            ai_response,
            history_messages=history_messages,
        ),
    )
```
## 2.6 流式对话编排

作用： 在已有同步业务逻辑的基础上实现 SSE 流式输出。

| 顺序  | 函数                         | 作用                            |
| --- | -------------------------- | ----------------------------- |
| 1   | `_build_hitl_event()`      | 提取 HITL 状态中前端所需的事件字段          |
| 2   | `chat_with_agent_stream()` | 完整的 SSE 流式对话入口                |
| 3   | `_agent_worker()`          | 流式入口中的内部函数，负责执行 Agent 并处理流式消息 |流式对话编排的总体流程与同步版本基本一致，主要区别是通过异步任务、消息队列和 SSE 实时输出执行过程：
流式对话编排的总体流程与同步版本基本一致，主要区别是通过异步任务、消息队列和 SSE 实时输出执行过程：
1. 先通过 SSE 发送初始进度，再与同步版本一样加载会话历史、长期记忆和待处理的 HITL 状态。
2. 创建带有 `asyncio.Queue` 的流式 `ChatRequestContext`，并与同步版本一样先保存用户消息。
3. 如果存在有效 `resume_state`，通过线程池恢复 RAG 并实时发送检索进度；否则通过 `agent.astream()` 和 `_agent_worker()` 流式执行 Agent。
4. 逐步发送 AI 回复 Token，并从 RAG 结果或 `ctx.take_rag_trace()` 获取执行轨迹，判断是否再次触发 HITL，向前端发送 Trace 和 HITL 事件。
5. 发送 SSE 结束标记 `[DONE]`，随后与同步版本一样更新 metadata、保存最终 AIMessage 和 RAG Trace，区别是长期记忆通过异步接口更新。

```python

def _build_hitl_event(pending_hitl: dict) -> dict:
    # 提取 HITL 状态中前端所需的事件字段
    return {
        "id": pending_hitl["id"],
        "prompt": pending_hitl["prompt"],
        "options": pending_hitl["options"],
        "route": pending_hitl["route"],
        "retrieval_status": pending_hitl["retrieval_status"],
        "original_question": pending_hitl["original_question"],
    }
    

async def chat_with_agent_stream(
    user_text: str,
    user_id: str = "default_user",
    session_id: str = "default_session",
):
    '''
    异步对话入口：负责从加载 Session 到保存最终回答的完整单轮编排。
    '''
    initial_step = {
        "type": "rag_step",# 前端根据 type 区分不同事件
        "step": {
            "icon": "📨",
            "label": "请求已接收，正在准备回答",
            "detail": "",
            "elapsed_ms": 0,
            "stage_elapsed_ms": 0,
        },
    } # SSE 流的初始进度通知，让前端在真正生成回答之前就能展示处理状态。
    # 同步版本通过return返回完整答案，异步版本使用 yield 持续发送 SSE 事件
    yield f"data: {json.dumps(initial_step)}\n\n"

    # 1. 从redis或数据库加载会话历史，判断是否需要 HITL 恢复
    messages, metadata = storage.load_with_meta(user_id, session_id)
    persistent_note = metadata.get("persistent_note", "")# 之前保存的长期记忆摘要，随着每轮对话滚动更新
    is_first_message = len(messages) == 0# 当前 Query 是否为这个 Session 的第一条消息
    # 从元数据里恢复待处理 HITL，识别无效/有效状态。
    stored_pending_hitl = metadata.get(PENDING_HITL_KEY)# 上一轮的hitl状态
    pending_hitl = _current_pending_hitl(stored_pending_hitl)# 校验
    invalid_pending_hitl = stored_pending_hitl is not None and pending_hitl is None # 数据库中有 HITL 状态，但校验失败
    is_hitl_resume = isinstance(pending_hitl, dict) # 本轮需要尝试恢复
    resume_state = _pending_resume_state(pending_hitl) # 提取 RAG Pipeline 恢复所需的快照
    # 如果是 HITL 补充，就组合原问题与补充信息
    effective_user_text = (
        _build_hitl_resume_query(pending_hitl, user_text) #usertext是hitl提示之后用户输入的补充信息
        if is_hitl_resume
        else user_text
    )
    hitl_answers = _existing_hitl_answers(pending_hitl)  # 人为补充信息，可能要多轮补充，因此列表累积usertext
    if is_hitl_resume: # 本轮也是一次补充，将其加入补充信息列表
        hitl_answers = [*hitl_answers, user_text]
    original_question = ( # 记录原始问题，供之后可能再次触发 HITL 时使用
        pending_hitl.get("original_question")
        if is_hitl_resume
        else user_text
    )

    # 2. 创建 流式ctx，先保存用户消息
    output_queue = asyncio.Queue() # 创建本轮请求专属的异步队列
    ctx = ChatRequestContext.for_stream( # 创建本次 Query 专属的同步运行上下文
        user_id=user_id,
        session_id=session_id,
        output_queue=output_queue,
    )
    ctx.reset_knowledge_tool_budget()

    try:
        messages.append(HumanMessage(content=user_text)) # 当前用户输入追加到会话历史
        storage.save(user_id, session_id, messages) # 保存完整消息列表

        # 3. 分支A: 上一轮 HITL 恢复
        if is_hitl_resume and resume_state:
            ## 3.1 获取当前正在运行的 asyncio 事件循环
            loop = asyncio.get_running_loop()
            ## 3.2 将同步 RAG 函数提交给线程池
            resume_future = loop.run_in_executor( #返回一个 asyncio.Future 对象，可以理解成未来会产生结果的任务凭证。
                None,
                lambda: _resume_rag_from_hitl_sync(pending_hitl, user_text, ctx),
            )

            ## 3.3 任务提交线程池之后，根据凭证不断判断RAG 没结束时，不断读取进度事件
            while not resume_future.done():
                try:
                    # outputqueue会通过ctx传递给rag，
                    # RAG Pipeline 内部如果调用：ctx.emit_rag_step("🔍", "正在检索知识库")
                    # 就可以通过 ctx 把进度事件送入 output_queue。
                    # 然后这里从队列中取出一条事件。如果队列为空，就等待。
                    event = await asyncio.wait_for(output_queue.get(), timeout=0.05)
                except asyncio.TimeoutError:
                    continue
                #将当前事件包装成 SSE 格式，交给前端。
                yield f"data: {json.dumps(event)}\n\n"

            ## 3.4 RAG 任务完成，不代表队列中的事件已经全部发送完毕。需要最后再把所有事件取出来
            while not output_queue.empty():
                event = output_queue.get_nowait()
                yield f"data: {json.dumps(event)}\n\n"

            ## 3.5 等待 resume_future 执行完成，并获取 RAG 函数的返回值。
            rag_result = await resume_future
            rag_trace = normalize_rag_trace(
                rag_result.get("rag_trace") if isinstance(rag_result, dict) else None
            )


            next_pending_hitl = None
            full_response = ""
            ## 3.6 恢复之后仍要补充信息
            if _is_hitl_trace(rag_trace):
                next_pending_hitl = _build_pending_hitl(
                    rag_trace,
                    original_question or user_text,
                    previous_answers=hitl_answers,
                    resume_state=rag_result.get("hitl_resume_state"),
                )
                full_response = _format_hitl_message(
                    next_pending_hitl["prompt"],
                    next_pending_hitl["options"],
                )
            ## 3.7 不需要 HITL，但没有检索到文档
            elif not (rag_result.get("docs") if isinstance(rag_result, dict) else None):
                full_response = _no_knowledge_response()
                yield f"data: {json.dumps({'type': 'content', 'content': full_response})}\n\n"
            ## 3.8 不需要 HITL，而且有检索证据，可以生成答案
            else:
                # 构建 LLM 输入：
                # SystemMessage + HumanMessage（原问题、补充信息、检索片段）
                answer_messages = _build_resume_answer_messages(
                    pending_hitl,
                    user_text,
                    rag_result.get("docs") or [],
                )
                # 使用 LLM 异步流式生成回答
                async for msg in model.astream(answer_messages):
                    # 累积完整回复，用于后续保存到数据库
                    content = _extract_ai_content(msg)
                    if content:
                        full_response += content
                        yield f"data: {json.dumps({'type': 'content', 'content': content})}\n\n"

            ## 3.9 将 RAG Trace 和 HITL 信息发送给前端
            if rag_trace:
                yield f"data: {json.dumps({'type': 'trace', 'rag_trace': rag_trace})}\n\n"

            if next_pending_hitl:
                yield f"data: {json.dumps({'type': 'hitl_request', 'hitl': _build_hitl_event(next_pending_hitl)})}\n\n"

            yield "data: [DONE]\n\n"

            ## 3.10 更新 metadata：标题、HITL、长期记忆
            save_meta = dict(metadata)
            if invalid_pending_hitl:# 清理无效的历史 HITL 状态
                save_meta[PENDING_HITL_KEY] = None
            if next_pending_hitl: #本轮再次触发 HITL，保存新的待恢复任务
                save_meta[PENDING_HITL_KEY] = next_pending_hitl
            else: # 本轮没有待处理 HITL
                save_meta[PENDING_HITL_KEY] = None # 清除上一轮待恢复的 HITL 状态
                # # 检查是否需要更新长期记忆
                if _should_update_persistent_note(messages, persistent_note):
                    try:
                        # 必须拿到更新后的长期记忆字符串，才能赋值
                        save_meta["persistent_note"] = await update_persistent_note(
                            persistent_note,
                            effective_user_text,
                            full_response,
                            history_messages=messages[:-1] if not persistent_note else None,
                        )
                    except Exception as e:
                        print(f"Update persistent note error: {e}")

            ## 3.11 保存 AI 消息和本轮 RAG Trace
            messages.append(AIMessage(content=full_response))
            extra_message_data = [None] * (len(messages) - 1) + [{"rag_trace": rag_trace}]
            storage.save(
                user_id,
                session_id,
                messages,
                metadata=save_meta,
                extra_message_data=extra_message_data,
            )
            return

        # 4. 分支B：普通对话，创建 Agent 并执行
        # 4.1 创建绑定当前 ctx 的 Agent
        request_agent = create_agent_for_request(ctx)
        ## 4.2 构建Agent输入上下文
        context_messages = _build_context_messages(messages[:-1], persistent_note, effective_user_text)

        ## 4.3 如果这是 Session 的第一条消息
        session_title = None
        if is_first_message:
            session_title = generate_session_title(user_text)
            yield f"data: {json.dumps({'type': 'session_title', 'title': session_title, 'session_id': session_id})}\n\n"

        full_response = ""
        agent_error = None

        async def _agent_worker():
            # 4.5 定义内部协程：负责执行 Agent，并将生成内容放进队列
            nonlocal full_response, agent_error
            try:
                 # 4.5.1 异步执行 Agent，持续获得消息块
                async for msg, _metadata in request_agent.astream(
                    {"messages": context_messages},
                    stream_mode="messages",
                    config={"recursion_limit": 8},
                ):
                     # 4.5.2 只接收 AIMessageChunk
                    if not isinstance(msg, AIMessageChunk):
                        continue
                    # 4.5.3 跳过工具调用参数块
                    if getattr(msg, "tool_call_chunks", None):
                        continue

                    # 4.5.4 从当前 AIMessageChunk 提取文本
                    content = ""
                    if isinstance(msg.content, str):
                        content = msg.content
                    elif isinstance(msg.content, list):
                        for block in msg.content:
                            if isinstance(block, str):
                                content += block
                            elif isinstance(block, dict) and block.get("type") == "text":
                                content += block.get("text", "")
    
                    # 4.5.5 只处理非空的文本块
                    if content:
                        # 查看当前是否已经触发 HITL
                        stored_trace = ctx.peek_rag_trace()
                        rag_trace = normalize_rag_trace(
                            stored_trace.get("rag_trace") if stored_trace else None
                        )
                        # 如果已经触发 HITL，
                        # 则不再将后续 Agent 文本当作普通回答
                        if _is_hitl_trace(rag_trace):
                            continue
                        # 累积完整答案，最终保存为 AIMessage
                        full_response += content
                        await output_queue.put({"type": "content", "content": content})
            except Exception as e:
                # 4.5.6 如果 Agent 执行失败，记录错误
                agent_error = str(e)
                # 将错误事件放入队列
                await output_queue.put({"type": "error", "content": str(e)})
            finally:
                await output_queue.put(None)

        # 4.6 将 _agent_worker() 调度为 asyncio Task
        # 任务会开始被事件循环调度，不需要等到 await 才启动
        agent_task = asyncio.create_task(_agent_worker())

        try:
            while True:# 4.7 主协程持续读取队列
                event = await output_queue.get()
                if event is None:
                    break
                yield f"data: {json.dumps(event)}\n\n"
        except GeneratorExit:
            # 异步生成器被关闭时，尝试取消 Agent Task
            agent_task.cancel()
            try:
                await agent_task
            except asyncio.CancelledError:
                pass
            raise
        finally:
            if not agent_task.done():
                agent_task.cancel()

        # 4.8 获取并清空当前请求上下文中的 RAG 信息
        stored_trace = ctx.take_rag_trace()
        rag_trace = normalize_rag_trace(stored_trace.get("rag_trace") if stored_trace else None)
        resume_state_from_trace = stored_trace.get("hitl_resume_state") if stored_trace else None
        next_pending_hitl = None
        hitl_response_content = ""
        if _is_hitl_trace(rag_trace):
            next_pending_hitl = _build_pending_hitl(
                rag_trace,
                original_question or user_text,
                previous_answers=hitl_answers,
                resume_state=resume_state_from_trace,
            )
            hitl_response_content = _format_hitl_message(
                next_pending_hitl["prompt"],
                next_pending_hitl["options"],
            )

        # 发送 Trace、HITL 和结束事件
        if rag_trace:
            yield f"data: {json.dumps({'type': 'trace', 'rag_trace': rag_trace})}\n\n"

        if next_pending_hitl:
            yield f"data: {json.dumps({'type': 'hitl_request', 'hitl': _build_hitl_event(next_pending_hitl)})}\n\n"

        yield "data: [DONE]\n\n"

        # 更新 metadata、长期记忆并保存消息
        save_meta = dict(metadata)
        if invalid_pending_hitl:
            save_meta[PENDING_HITL_KEY] = None
        if session_title:
            save_meta["title"] = session_title

        if next_pending_hitl:
            save_meta[PENDING_HITL_KEY] = next_pending_hitl
            full_response = hitl_response_content
        else:
            if is_hitl_resume and not agent_error:
                save_meta[PENDING_HITL_KEY] = None
            if _should_update_persistent_note(messages, persistent_note):
                try:
                    save_meta["persistent_note"] = await update_persistent_note(
                        persistent_note,
                        effective_user_text,
                        full_response,
                        history_messages=messages[:-1] if not persistent_note else None,
                    )
                except Exception as e:
                    print(f"Update persistent note error: {e}")

        messages.append(AIMessage(content=full_response))
        extra_message_data = [None] * (len(messages) - 1) + [{"rag_trace": rag_trace}]
        storage.save(
            user_id,
            session_id,
            messages,
            metadata=save_meta,
            extra_message_data=extra_message_data,
        )
    finally:
        ctx.close()
```
